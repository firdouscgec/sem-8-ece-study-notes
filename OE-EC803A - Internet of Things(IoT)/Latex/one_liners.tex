\section*{Chapter 1: Introduction to IoT}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Internet of Things (IoT) definition}} $\quad\to\quad$ Global network of self-configuring physical and virtual objects.
  \item \textcolor{darkblue}{\textbf{World Wide Web vs. IoT core distinction}} $\quad\to\quad$ WWW connects humans; IoT connects physical and virtual devices.
  \item \textcolor{darkblue}{\textbf{Enchanted Object definition}} $\quad\to\quad$ Everyday physical object enhanced with ambient connectivity and sensors.
  \item \textcolor{darkblue}{\textbf{Localized network focused purely on data collection}} $\quad\to\quad$ Wireless Sensor Network (WSN).
  \item \textcolor{darkblue}{\textbf{Traditional Machine-to-Machine (M2M) communication style}} $\quad\to\quad$ Closed, point-to-point, isolated, and hardware-centric.
  \item \textcolor{darkblue}{\textbf{Direct local device communication model}} $\quad\to\quad$ Device-to-Device (D2D).
  \item \textcolor{darkblue}{\textbf{Direct internet platform connection model}} $\quad\to\quad$ Device-to-Cloud (D2C).
  \item \textcolor{darkblue}{\textbf{IoT Gateway primary function}} $\quad\to\quad$ Performs protocol translation and data routing.
  \item \textcolor{darkblue}{\textbf{ETSI M2M architecture interfaces}} $\quad\to\quad$ dIa (device-app), mIa (network-app), mId (M2M-device).
  \item \textcolor{darkblue}{\textbf{Generic M2M system solution layers}} $\quad\to\quad$ M2M Area Network, Communication Network, M2M Applications.
  \item \textcolor{darkblue}{\textbf{Four stages of IoT information value chain}} $\quad\to\quad$ Sense, Transmit, Store/Analyze, Decide/Act.
  \item \textcolor{darkblue}{\textbf{OGC API standard for querying sensor observations}} $\quad\to\quad$ SOS (Sensor Observation Service).
  \item \textcolor{darkblue}{\textbf{OGC XML schema for sensor parameters}} $\quad\to\quad$ SensorML (Sensor Model Language).
  \item \textcolor{darkblue}{\textbf{Category used for business-to-consumer processes}} $\quad\to\quad$ Personal IoT.
  \item \textcolor{darkblue}{\textbf{Category used by citizens to benefit smart cities}} $\quad\to\quad$ Community IoT.
  \item \textcolor{darkblue}{\textbf{Automatic device communication without human intervention}} $\quad\to\quad$ Machine-to-Machine (M2M).
  \item \textcolor{darkblue}{\textbf{Network of uniquely identifiable embedded computing devices}} $\quad\to\quad$ Internet of Things (IoT).
  \item \textcolor{darkblue}{\textbf{Core components providing physical world awareness}} $\quad\to\quad$ Sensors.
  \item \textcolor{darkblue}{\textbf{Vital elements required by house key computers}} $\quad\to\quad$ Data and services.
  \item \textcolor{darkblue}{\textbf{M2M device domain alternative name}} $\quad\to\quad$ M2M Area Network.
  \item \textcolor{darkblue}{\textbf{Industrial IoT (IIoT) priority focus}} $\quad\to\quad$ Zero-downtime, safety, and mission-critical reliability.
\end{enumerate}

\section*{Chapter 2: Design Principles for Connected Devices}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Calm Technology primary goal}} $\quad\to\quad$ Blends into the periphery, reducing cognitive fatigue.
  \item \textcolor{darkblue}{\textbf{Calm Technology design rules}} $\quad\to\quad$ Peripheral attention, seamless shift, presence, reach, attention limits.
  \item \textcolor{darkblue}{\textbf{Ambient Technology definition}} $\quad\to\quad$ Devices displaying status information via subtle environmental cues.
  \item \textcolor{darkblue}{\textbf{Ambient Orb designer}} $\quad\to\quad$ David Rose.
  \item \textcolor{darkblue}{\textbf{Ambient Orb feedback mechanism}} $\quad\to\quad$ Frosted sphere changing color using internal LEDs.
  \item \textcolor{darkblue}{\textbf{Ambient Orb main limitation}} $\quad\to\quad$ Low information density (conveys status, not text).
  \item \textcolor{darkblue}{\textbf{Magic as Metaphor concept}} $\quad\to\quad$ Using mythical archetypes to explain wireless operations.
  \item \textcolor{darkblue}{\textbf{Gesture-controlled magical controller device example}} $\quad\to\quad$ Kymera Magic Wand.
  \item \textcolor{darkblue}{\textbf{Critiques of Magic as Metaphor}} $\quad\to\quad$ Expectation mismatch, lack of debugging/error states.
  \item \textcolor{darkblue}{\textbf{Privacy design solution for local devices}} $\quad\to\quad$ Edge processing and data minimization.
  \item \textcolor{darkblue}{\textbf{Web Thinking for Connected Devices definition}} $\quad\to\quad$ Building IoT architectures using standardized HTTP/REST principles.
  \item \textcolor{darkblue}{\textbf{URI role in Web Thinking}} $\quad\to\quad$ Unique address identifier for every physical sensor/actuator.
  \item \textcolor{darkblue}{\textbf{REST command to read sensor state}} $\quad\to\quad$ GET.
  \item \textcolor{darkblue}{\textbf{REST command to trigger actuator action}} $\quad\to\quad$ POST.
  \item \textcolor{darkblue}{\textbf{Physical Mashup definition}} $\quad\to\quad$ Linking different physical devices using standard web APIs.
  \item \textcolor{darkblue}{\textbf{Affordance definition}} $\quad\to\quad$ Physical properties of an object suggesting its usage.
  \item \textcolor{darkblue}{\textbf{Clash of Affordances in IoT}} $\quad\to\quad$ Traditional appearance hiding invisible digital sensor interfaces.
  \item \textcolor{darkblue}{\textbf{Visual signifiers resolving Clash of Affordances}} $\quad\to\quad$ LED rings or textured borders guiding interaction.
  \item \textcolor{darkblue}{\textbf{Graceful degradation design requirement}} $\quad\to\quad$ Device remains functional physically if internet or power fails.
  \item \textcolor{darkblue}{\textbf{Ambient device pulsing when rain is expected}} $\quad\to\quad$ Ambient Umbrella.
\end{enumerate}

\section*{Chapter 3: Internet Principles}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{BLE full form}} $\quad\to\quad$ Bluetooth Low Energy.
  \item \textcolor{darkblue}{\textbf{BLE GATT hierarchy data structures}} $\quad\to\quad$ Profiles, Services, and Characteristics.
  \item \textcolor{darkblue}{\textbf{Classic Bluetooth vs. BLE latency}} $\quad\to\quad$ Classic up to 6s; BLE under 3ms.
  \item \textcolor{darkblue}{\textbf{Address style permanent and manually configured}} $\quad\to\quad$ Static IP.
  \item \textcolor{darkblue}{\textbf{DHCP dynamic configuration handshake acronym}} $\quad\to\quad$ DORA.
  \item \textcolor{darkblue}{\textbf{DORA handshake steps}} $\quad\to\quad$ Discover, Offer, Request, Acknowledge.
  \item \textcolor{darkblue}{\textbf{TCP full form}} $\quad\to\quad$ Transmission Control Protocol.
  \item \textcolor{darkblue}{\textbf{UDP full form}} $\quad\to\quad$ User Datagram Protocol.
  \item \textcolor{darkblue}{\textbf{Reliable, connection-oriented transport protocol}} $\quad\to\quad$ TCP.
  \item \textcolor{darkblue}{\textbf{Fast, lightweight, connectionless transport protocol}} $\quad\to\quad$ UDP.
  \item \textcolor{darkblue}{\textbf{UDP minimum header size}} $\quad\to\quad$ 8 bytes.
  \item \textcolor{darkblue}{\textbf{TCP minimum header size}} $\quad\to\quad$ 20 bytes.
  \item \textcolor{darkblue}{\textbf{IPv6 address length in bits}} $\quad\to\quad$ 128 bits.
  \item \textcolor{darkblue}{\textbf{IPv4 address length in bits}} $\quad\to\quad$ 32 bits.
  \item \textcolor{darkblue}{\textbf{IPv6 feature for stateless autoconfiguration}} $\quad\to\quad$ SLAAC.
  \item \textcolor{darkblue}{\textbf{MAC address length in bits}} $\quad\to\quad$ 48 bits.
  \item \textcolor{darkblue}{\textbf{MAC address data link layer level}} $\quad\to\quad$ Layer 2 (Physical address).
  \item \textcolor{darkblue}{\textbf{MAC address manufacturer identifier first 24 bits}} $\quad\to\quad$ OUI (Organizationally Unique Identifier).
  \item \textcolor{darkblue}{\textbf{Logical endpoints routing network traffic in OS}} $\quad\to\quad$ Network Ports (16-bit).
  \item \textcolor{darkblue}{\textbf{HTTP message format style}} $\quad\to\quad$ ASCII text headers.
  \item \textcolor{darkblue}{\textbf{CoAP transport protocol and minimum header size}} $\quad\to\quad$ UDP; 4 bytes.
  \item \textcolor{darkblue}{\textbf{MQTT architecture model and minimum header size}} $\quad\to\quad$ Publish-Subscribe (Broker-based); 2 bytes.
  \item \textcolor{darkblue}{\textbf{Weightless protocol radio spectrum utilization}} $\quad\to\quad$ Sub-GHz TV White Space (TVWS).
  \item \textcolor{darkblue}{\textbf{Computing proximity closest to data source}} $\quad\to\quad$ Mist computing.
  \item \textcolor{darkblue}{\textbf{Proximity order of computing layers}} $\quad\to\quad$ Mist → Edge → Fog → Cloud.
  \item \textcolor{darkblue}{\textbf{Computing layer running on regional routers}} $\quad\to\quad$ Fog computing.
  \item \textcolor{darkblue}{\textbf{Layer connecting IoT devices with WAN}} $\quad\to\quad$ Network layer.
  \item \textcolor{darkblue}{\textbf{IoTA full form}} $\quad\to\quad$ Internet of Things Architecture.
  \item \textcolor{darkblue}{\textbf{Protocol linking all devices in the internet}} $\quad\to\quad$ TCP/IP.
  \item \textcolor{darkblue}{\textbf{Computing emphasizing proximity to end user}} $\quad\to\quad$ Fog computing.
\end{enumerate}

\section*{Chapter 8: Techniques for Writing Embedded Code}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Primary risk of dynamic memory allocation in MCU}} $\quad\to\quad$ Heap fragmentation leading to crashes.
  \item \textcolor{darkblue}{\textbf{Dynamic memory allocation alternative}} $\quad\to\quad$ Static allocation of fixed-size buffers.
  \item \textcolor{darkblue}{\textbf{Blocking delay side effect}} $\quad\to\quad$ Halts CPU cycles, blocking event handlers.
  \item \textcolor{darkblue}{\textbf{Non-blocking delay implementation}} $\quad\to\quad$ State machines using `millis()` checks.
  \item \textcolor{darkblue}{\textbf{Battery life optimization technique using low power}} $\quad\to\quad$ CPU deep sleep modes.
  \item \textcolor{darkblue}{\textbf{Action to audit third-party libraries}} $\quad\to\quad$ Checking Flash and RAM consumption reports.
  \item \textcolor{darkblue}{\textbf{Serial print debugging drawback}} $\quad\to\quad$ Alters timing execution and consumes RAM/UART.
  \item \textcolor{darkblue}{\textbf{JTAG/SWD debugging advantage}} $\quad\to\quad$ Non-intrusive breakpoints and real-time register audit.
  \item \textcolor{darkblue}{\textbf{Reliability phase forwarding packets hop-by-hop}} $\quad\to\quad$ Message relaying.
  \item \textcolor{darkblue}{\textbf{Reliability phase tracking sequence counter gaps}} $\quad\to\quad$ Lost message detection.
  \item \textcolor{darkblue}{\textbf{Reliability phase asking for missing packets only}} $\quad\to\quad$ Selective recovery.
  \item \textcolor{darkblue}{\textbf{Reliability first step in WSN reliable transfer}} $\quad\to\quad$ Initialization.
  \item \textcolor{darkblue}{\textbf{CPU clock reduction scaling method}} $\quad\to\quad$ DVFS (Dynamic Voltage and Frequency Scaling).
  \item \textcolor{darkblue}{\textbf{Non-intrusive debugging probe connector}} $\quad\to\quad$ JTAG or SWD.
  \item \textcolor{darkblue}{\textbf{Best storage for persistent configuration settings}} $\quad\to\quad$ EEPROM or Flash.
\end{enumerate}
