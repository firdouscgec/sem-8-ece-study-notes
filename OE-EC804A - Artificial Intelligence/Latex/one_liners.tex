\section*{Chapter 1: Introduction to Artificial Intelligence}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Artificial Intelligence definition}} $\quad\to\quad$ Branch of computer science building smart, autonomous systems.
  \item \textcolor{darkblue}{\textbf{Standard modern approach to AI}} $\quad\to\quad$ Acting Rationally (Rational Agent approach).
  \item \textcolor{darkblue}{\textbf{Two dimensions of conceptual approaches to AI}} $\quad\to\quad$ Thought processes vs. behavior; human-like vs. rational.
  \item \textcolor{darkblue}{\textbf{Thinking Humanly focus}} $\quad\to\quad$ Mimicking cognitive activities and mental processes of humans.
  \item \textcolor{darkblue}{\textbf{Thinking Rationally focus}} $\quad\to\quad$ Using formal logic and laws of thought.
  \item \textcolor{darkblue}{\textbf{Acting Humanly focus}} $\quad\to\quad$ Performing actions indistinguishable from a human.
  \item \textcolor{darkblue}{\textbf{Acting Rationally focus}} $\quad\to\quad$ Maximizing expected performance or payoff.
  \item \textcolor{darkblue}{\textbf{Inventor of the Turing Test}} $\quad\to\quad$ Alan Turing.
  \item \textcolor{darkblue}{\textbf{Year Turing Test was introduced}} $\quad\to\quad$ 1950.
  \item \textcolor{darkblue}{\textbf{Original name of the Turing Test}} $\quad\to\quad$ The Imitation Game.
  \item \textcolor{darkblue}{\textbf{Duration of interrogation in Turing Test}} $\quad\to\quad$ 5 minutes.
  \item \textcolor{darkblue}{\textbf{NLP purpose in Turing Test}} $\quad\to\quad$ To communicate successfully in natural language.
  \item \textcolor{darkblue}{\textbf{Knowledge Representation purpose in Turing Test}} $\quad\to\quad$ To store information acquired before or during conversation.
  \item \textcolor{darkblue}{\textbf{Automated Reasoning purpose in Turing Test}} $\quad\to\quad$ To draw conclusions from stored information.
  \item \textcolor{darkblue}{\textbf{Machine Learning purpose in Turing Test}} $\quad\to\quad$ To adapt to new circumstances and detect patterns.
  \item \textcolor{darkblue}{\textbf{Two additional requirements for Total Turing Test}} $\quad\to\quad$ Computer Vision and Robotics.
  \item \textcolor{darkblue}{\textbf{Cognitive Science discipline contribution to AI}} $\quad\to\quad$ Mimicking human cognitive processes.
  \item \textcolor{darkblue}{\textbf{Philosophy contribution to AI}} $\quad\to\quad$ Rules of logic and mind-machine concepts.
  \item \textcolor{darkblue}{\textbf{Mathematics contribution to AI}} $\quad\to\quad$ Logic, probability, algorithms, and tractability.
  \item \textcolor{darkblue}{\textbf{Economics contribution to AI}} $\quad\to\quad$ Utility theory, decision theory, and game theory.
  \item \textcolor{darkblue}{\textbf{Neuroscience contribution to AI}} $\quad\to\quad$ Biological models of information processing (ANNs).
  \item \textcolor{darkblue}{\textbf{Psychology contribution to AI}} $\quad\to\quad$ Behavioral and cognitive models of human action.
  \item \textcolor{darkblue}{\textbf{Control Theory contribution to AI}} $\quad\to\quad$ Feedback loops and self-regulating systems.
  \item \textcolor{darkblue}{\textbf{Computer Vision main function}} $\quad\to\quad$ Extracting structured information from digital images or videos.
  \item \textcolor{darkblue}{\textbf{Natural Language Processing main function}} $\quad\to\quad$ Processing and analyzing human languages.
  \item \textcolor{darkblue}{\textbf{Machine Learning main function}} $\quad\to\quad$ Improving task performance automatically from training data.
  \item \textcolor{darkblue}{\textbf{Expert System definition}} $\quad\to\quad$ Software emulating human expert decision-making in a narrow domain.
  \item \textcolor{darkblue}{\textbf{Three main components of an Expert System}} $\quad\to\quad$ Knowledge Base, Inference Engine, and User Interface.
  \item \textcolor{darkblue}{\textbf{Expert System Knowledge Base storage format}} $\quad\to\quad$ IF-THEN production rules.
  \item \textcolor{darkblue}{\textbf{Forward Chaining search bias}} $\quad\to\quad$ Data-driven reasoning from facts to conclusions.
  \item \textcolor{darkblue}{\textbf{Backward Chaining search bias}} $\quad\to\quad$ Goal-driven reasoning starting from hypothesis to facts.

\section*{Chapter 2: Intelligent Agents}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{AI Agent definition}} $\quad\to\quad$ Anything perceiving environment through sensors and acting via actuators.
  \item \textcolor{darkblue}{\textbf{Percept Sequence definition}} $\quad\to\quad$ Complete history of all percepts received to date.
  \item \textcolor{darkblue}{\textbf{Agent Function mathematical mapping}} $\quad\to\quad$ $f: P^* \rightarrow A$
  \item \textcolor{darkblue}{\textbf{Agent Program definition}} $\quad\to\quad$ Concrete implementation of agent function on physical architecture.
  \item \textcolor{darkblue}{\textbf{Agent formula}} $\quad\to\quad$ $\text{Agent} = \text{Architecture} + \text{Program}$
  \item \textcolor{darkblue}{\textbf{PEAS acronym components}} $\quad\to\quad$ Performance, Environment, Actuators, Sensors.
  \item \textcolor{darkblue}{\textbf{PEAS Performance Measure for Automated Taxi}} $\quad\to\quad$ Safe arrival, passenger comfort, legal driving, and profit.
  \item \textcolor{darkblue}{\textbf{PEAS Environment for Automated Taxi}} $\quad\to\quad$ Roads, traffic, pedestrians, weather, and customers.
  \item \textcolor{darkblue}{\textbf{PEAS Actuators for Automated Taxi}} $\quad\to\quad$ Steering wheel, accelerator, brakes, horn, and display.
  \item \textcolor{darkblue}{\textbf{PEAS Sensors for Automated Taxi}} $\quad\to\quad$ Cameras, LIDAR, sonar, GPS, and speedometer.
  \item \textcolor{darkblue}{\textbf{PEAS Sensors for Medical Diagnosis System}} $\quad\to\quad$ Keyboard or voice input of symptoms.
  \item \textcolor{darkblue}{\textbf{Autonomy in agents}} $\quad\to\quad$ Behavior guided by own experience, not built-in designer knowledge.
  \item \textcolor{darkblue}{\textbf{Five basic agent structures}} $\quad\to\quad$ Simple reflex, model-based, goal-based, utility-based, learning agents.
  \item \textcolor{darkblue}{\textbf{Simple Reflex Agent action driver}} $\quad\to\quad$ Current percept only, using condition-action rules.
  \item \textcolor{darkblue}{\textbf{Key limitation of Simple Reflex Agents}} $\quad\to\quad$ Requires a fully observable environment.
  \item \textcolor{darkblue}{\textbf{Model-Based Reflex Agent key feature}} $\quad\to\quad$ Maintains internal state to track unobserved world aspects.
  \item \textcolor{darkblue}{\textbf{Goal-Based Agent action selector}} $\quad\to\quad$ Chooses actions that achieve explicit goal states.
  \item \textcolor{darkblue}{\textbf{Utility-Based Agent action selector}} $\quad\to\quad$ Maximizes a real-valued utility function indicating state preference.
  \item \textcolor{darkblue}{\textbf{Four components of a Learning Agent}} $\quad\to\quad$ Learning element, performance element, critic, problem generator.
  \item \textcolor{darkblue}{\textbf{Learning Element function}} $\quad\to\quad$ Making improvements by learning from experience.
  \item \textcolor{darkblue}{\textbf{Performance Element function}} $\quad\to\quad$ Selecting external actions to execute.
  \item \textcolor{darkblue}{\textbf{Critic function}} $\quad\to\quad$ Evaluating behavior against external performance standards.
  \item \textcolor{darkblue}{\textbf{Problem Generator function}} $\quad\to\quad$ Suggesting exploratory actions to discover new knowledge.
  \item \textcolor{darkblue}{\textbf{Fully Observable environment}} $\quad\to\quad$ Sensors detect complete state of environment at each point.
  \item \textcolor{darkblue}{\textbf{Partially Observable environment}} $\quad\to\quad$ Sensors have noise or gaps leaving state details unknown.
  \item \textcolor{darkblue}{\textbf{Deterministic environment}} $\quad\to\quad$ Next state completely determined by current state and action.
  \item \textcolor{darkblue}{\textbf{Stochastic environment}} $\quad\to\quad$ Next state contains random uncertainty.
  \item \textcolor{darkblue}{\textbf{Strategic environment}} $\quad\to\quad$ Uncertainty arises only from actions of other agents.
  \item \textcolor{darkblue}{\textbf{Episodic environment}} $\quad\to\quad$ Independent, atomic episodes where past actions don't affect future.
  \item \textcolor{darkblue}{\textbf{Sequential environment}} $\quad\to\quad$ Current decisions affect all future states and actions.
  \item \textcolor{darkblue}{\textbf{Static environment}} $\quad\to\quad$ Environment does not change during agent decision-making.
  \item \textcolor{darkblue}{\textbf{Dynamic environment}} $\quad\to\quad$ Environment changes while agent is deciding.
  \item \textcolor{darkblue}{\textbf{Semidynamic environment}} $\quad\to\quad$ Environment does not change, but performance score decreases with time.
  \item \textcolor{darkblue}{\textbf{Discrete environment}} $\quad\to\quad$ Finite or countable states, percepts, and actions.
  \item \textcolor{darkblue}{\textbf{Continuous environment}} $\quad\to\quad$ Continuous real variables represent environment states.
  \item \textcolor{darkblue}{\textbf{Chess with clock environment properties}} $\quad\to\quad$ Fully observable, strategic, sequential, semidynamic, discrete, multi-agent.
  \item \textcolor{darkblue}{\textbf{Taxi driving environment properties}} $\quad\to\quad$ Partially observable, stochastic, sequential, dynamic, continuous, multi-agent.
\end{enumerate}

\section*{Chapter 3: Solving Problems by Searching (Uninformed Search)}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Problem-Solving Agent definition}} $\quad\to\quad$ Goal-based agent finding action sequences leading to goal states.
  \item \textcolor{darkblue}{\textbf{Five components of problem formulation}} $\quad\to\quad$ Initial state, actions, transition model, goal test, path cost.
  \item \textcolor{darkblue}{\textbf{Transition Model definition}} $\quad\to\quad$ Function $Result(s, a)$ returning state reached by action.
  \item \textcolor{darkblue}{\textbf{Path Cost formula}} $\quad\to\quad$ $g(n) = \text{sum of step costs along path}$.
  \item \textcolor{darkblue}{\textbf{State Space representation}} $\quad\to\quad$ Directed graph with states as nodes and actions as edges.
  \item \textcolor{darkblue}{\textbf{Four search evaluation criteria}} $\quad\to\quad$ Completeness, time complexity, space complexity, optimality.
  \item \textcolor{darkblue}{\textbf{Branching factor ($b$)}} $\quad\to\quad$ Maximum number of successors of any node.
  \item \textcolor{darkblue}{\textbf{Uninformed Search definition}} $\quad\to\quad$ Search using only problem definition without heuristic guidance.
  \item \textcolor{darkblue}{\textbf{BFS search strategy}} $\quad\to\quad$ Explores level-by-level, expanding all shallowest nodes first.
  \item \textcolor{darkblue}{\textbf{BFS frontier data structure}} $\quad\to\quad$ FIFO Queue (First-In, First-Out).
  \item \textcolor{darkblue}{\textbf{BFS Space Complexity}} $\quad\to\quad$ $O(b^d)$ (Exponential memory).
  \item \textcolor{darkblue}{\textbf{BFS Optimality condition}} $\quad\to\quad$ Step costs must be uniform.
  \item \textcolor{darkblue}{\textbf{DFS search strategy}} $\quad\to\quad$ Explores deep into a branch before backtracking.
  \item \textcolor{darkblue}{\textbf{DFS frontier data structure}} $\quad\to\quad$ LIFO Stack or Recursion.
  \item \textcolor{darkblue}{\textbf{DFS Space Complexity}} $\quad\to\quad$ $O(bm)$ (Linear memory).
  \item \textcolor{darkblue}{\textbf{Reason DFS is not complete}} $\quad\to\quad$ Can get trapped in infinite paths or loops.
  \item \textcolor{darkblue}{\textbf{DLS definition}} $\quad\to\quad$ DFS running with a pre-specified depth limit $l$.
  \item \textcolor{darkblue}{\textbf{IDDFS definition}} $\quad\to\quad$ Repeated DLS with increasing limits ($l = 0, 1, 2...$).
  \item \textcolor{darkblue}{\textbf{IDDFS Space Complexity}} $\quad\to\quad$ $O(bd)$ (Linear space).
  \item \textcolor{darkblue}{\textbf{IDDFS Completeness status}} $\quad\to\quad$ Complete (if $b$ is finite).
  \item \textcolor{darkblue}{\textbf{UCS node expansion criteria}} $\quad\to\quad$ Node with lowest accumulated path cost $g(n)$.
  \item \textcolor{darkblue}{\textbf{UCS frontier data structure}} $\quad\to\quad$ Priority Queue.
  \item \textcolor{darkblue}{\textbf{UCS Time and Space Complexity}} $\quad\to\quad$ $O(b^{1 + \lfloor C^*/\epsilon \rfloor})$.
  \item \textcolor{darkblue}{\textbf{UCS completeness condition}} $\quad\to\quad$ Step costs must be positive ($\ge \epsilon > 0$).
  \item \textcolor{darkblue}{\textbf{8-Puzzle Initial State}} $\quad\to\quad$ Scrambled configuration of 8 numbered tiles in $3\times3$ grid.
  \item \textcolor{darkblue}{\textbf{8-Puzzle Actions}} $\quad\to\quad$ Moving blank space Left, Right, Up, or Down.
  \item \textcolor{darkblue}{\textbf{Missionaries and Cannibals safety constraint}} $\quad\to\quad$ Cannibals must never outnumber missionaries on either bank.
  \item \textcolor{darkblue}{\textbf{Missionaries and Cannibals State representation}} $\quad\to\quad$ $(M, C, B)$ for Left bank values.
  \item \textcolor{darkblue}{\textbf{Goal state in River Crossing problem}} $\quad\to\quad$ $(0, 0, 0)$.
  \item \textcolor{darkblue}{\textbf{Water Jug Problem initial state}} $\quad\to\quad$ $(0, 0)$ (both jugs empty).
  \item \textcolor{darkblue}{\textbf{Water Jug Problem Goal State}} $\quad\to\quad$ $(2, y)$ where 4-gallon jug has 2 gallons.
  \item \textcolor{darkblue}{\textbf{Water Jug rule for filling 3-Gal jug}} $\quad\to\quad$ $(x, 3)$ if $y < 3$.
  \item \textcolor{darkblue}{\textbf{Water Jug rule for pouring 4-Gal into 3-Gal until full}} $\quad\to\quad$ $(x - (3 - y), 3)$ if $x+y \ge 3$.
  \item \textcolor{darkblue}{\textbf{Water Jug rule for pouring all 3-Gal into 4-Gal}} $\quad\to\quad$ $(x+y, 0)$ if $x+y \le 4$.
\end{enumerate}

\section*{Chapter 4: Informed Search \& Exploration}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Informed Search definition}} $\quad\to\quad$ Search using problem-specific heuristic knowledge to guide pathfinding.
  \item \textcolor{darkblue}{\textbf{Heuristic function $h(n)$}} $\quad\to\quad$ Estimated cost of cheapest path from $n$ to goal.
  \item \textcolor{darkblue}{\textbf{Greedy Best-First Search evaluation function}} $\quad\to\quad$ $f(n)=h(n)$.
  \item \textcolor{darkblue}{\textbf{Greedy Best-First Search expansion bias}} $\quad\to\quad$ Expands node closest to goal.
  \item \textcolor{darkblue}{\textbf{Greedy Best-First Search space complexity}} $\quad\to\quad$ $O(b^l)$ (Exponential space).
  \item \textcolor{darkblue}{\textbf{A* Search evaluation function}} $\quad\to\quad$ $f(n)=g(n)+h(n)$.
  \item \textcolor{darkblue}{\textbf{A* Search completeness condition}} $\quad\to\quad$ Branching factor finite; step costs $\ge\epsilon>0$.
  \item \textcolor{darkblue}{\textbf{Admissible heuristic definition}} $\quad\to\quad$ Heuristic never overestimating true cost to goal ($0\le h(n)\le h^*(n)$).
  \item \textcolor{darkblue}{\textbf{Consistent (monotone) heuristic definition}} $\quad\to\quad$ $h(n)\le c(n,a,n')+h(n')$ (triangle inequality).
  \item \textcolor{darkblue}{\textbf{A* tree search optimality condition}} $\quad\to\quad$ Heuristic $h(n)$ must be admissible.
  \item \textcolor{darkblue}{\textbf{A* graph search optimality condition}} $\quad\to\quad$ Heuristic $h(n)$ must be consistent.
  \item \textcolor{darkblue}{\textbf{Relation between consistency and admissibility}} $\quad\to\quad$ Every consistent heuristic is admissible.
  \item \textcolor{darkblue}{\textbf{Monotonicity of $f$-values in A*}} $\quad\to\quad$ If $h(n)$ is consistent, $f(n)$ is non-decreasing along paths.
  \item \textcolor{darkblue}{\textbf{Manhattan Distance heuristic formula for 8-Puzzle}} $\quad\to\quad$ Sum of horizontal and vertical distances from goal.
  \item \textcolor{darkblue}{\textbf{Misplaced Tiles heuristic definition}} $\quad\to\quad$ Count of tiles not in their goal position.
  \item \textcolor{darkblue}{\textbf{Simulated Annealing working principle}} $\quad\to\quad$ Local search accepting bad moves with decreasing probability over time.
  \item \textcolor{darkblue}{\textbf{Simulated Annealing acceptance probability formula}} $\quad\to\quad$ $P = e^{\frac{\Delta E}{T}}$.
  \item \textcolor{darkblue}{\textbf{Impact of slow cooling in Simulated Annealing}} $\quad\to\quad$ Guaranteed to find global optimum with probability 1.
  \item \textcolor{darkblue}{\textbf{Impact of fast cooling in Simulated Annealing}} $\quad\to\quad$ Traps system in a local optimum.
  \item \textcolor{darkblue}{\textbf{Hill-Climbing Search definition}} $\quad\to\quad$ Greedy local search moving to neighbor with highest value.
  \item \textcolor{darkblue}{\textbf{Three limitations of Hill-Climbing}} $\quad\to\quad$ Local Maxima, Ridges, Plateaus.
  \item \textcolor{darkblue}{\textbf{Flat local maximum vs. Shoulder on a Plateau}} $\quad\to\quad$ Flat maximum has no exit; shoulder allows progression.
  \item \textcolor{darkblue}{\textbf{Solution to Hill-Climbing limitations}} $\quad\to\quad$ Random-Restart Hill Climbing.
  \item \textcolor{darkblue}{\textbf{Local Beam Search working principle}} $\quad\to\quad$ Tracks $k$ states in parallel, keeping best $k$ successors.
  \item \textcolor{darkblue}{\textbf{AO* Search application}} $\quad\to\quad$ Finding solutions in AND-OR graphs (decomposable problems).
  \item \textcolor{darkblue}{\textbf{AO* evaluation function for AND branch}} $\quad\to\quad$ $f(n) = g(n) + \sum h(\text{sub-problems})$.
  \item \textcolor{darkblue}{\textbf{Recursive Best-First Search (RBFS) Space Complexity}} $\quad\to\quad$ $O(bd)$ (Linear space).
  \item \textcolor{darkblue}{\textbf{RBFS backtracking trigger}} $\quad\to\quad$ When current node $f$-value exceeds the $f$-limit.
  \item \textcolor{darkblue}{\textbf{RBFS back-propagation mechanism}} $\quad\to\quad$ Replaces parent's $f$-value with best child's $f$-value on backtrack.
  \item \textcolor{darkblue}{\textbf{RBFS time complexity limitation}} $\quad\to\quad$ Exponential time due to node regeneration.
\end{enumerate}

  \item \textcolor{darkblue}{\textbf{Arc consistency (AC-3) definition}} $\quad\to\quad$ For every $X$ value, some $Y$ value is allowed.
\end{enumerate}

\section*{Chapter 6: Adversarial Search}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Minimax definition}} $\quad\to\quad$ Recursive decision algorithm for two-player, zero-sum, perfect-information games.
  \item \textcolor{darkblue}{\textbf{Zero-sum game definition}} $\quad\to\quad$ Opponent's gain is exactly equal to player's loss.
  \item \textcolor{darkblue}{\textbf{Player MAX objective in Minimax}} $\quad\to\quad$ Maximize the final utility score.
  \item \textcolor{darkblue}{\textbf{Player MIN objective in Minimax}} $\quad\to\quad$ Minimize the MAX utility score.
  \item \textcolor{darkblue}{\textbf{Value propagated from child at a MAX node}} $\quad\to\quad$ Maximum of the child values.
  \item \textcolor{darkblue}{\textbf{Value propagated from child at a MIN node}} $\quad\to\quad$ Minimum of the children's values.
  \item \textcolor{darkblue}{\textbf{Minimax Time Complexity}} $\quad\to\quad$ $O(b^d)$ (Exponential).
  \item \textcolor{darkblue}{\textbf{Minimax Space Complexity}} $\quad\to\quad$ $O(bd)$ (Linear space).
  \item \textcolor{darkblue}{\textbf{Alpha-Beta Pruning definition}} $\quad\to\quad$ Optimization skipping subtrees that won't change minimax decision.
  \item \textcolor{darkblue}{\textbf{Alpha ($\alpha$) in pruning}} $\quad\to\quad$ Best (highest) choice found so far for MAX.
  \item \textcolor{darkblue}{\textbf{Beta ($\beta$) in pruning}} $\quad\to\quad$ Best (lowest) choice found so far for MIN.
  \item \textcolor{darkblue}{\textbf{Initial values of Alpha and Beta}} $\quad\to\quad$ $\alpha = -\infty$ and $\beta = +\infty$.
  \item \textcolor{darkblue}{\textbf{Pruning cutoff condition}} $\quad\to\quad$ When $\beta \le \alpha$.
  \item \textcolor{darkblue}{\textbf{Alpha-cutoff location}} $\quad\to\quad$ Occurs at a MIN node when $\beta \le \alpha$.
  \item \textcolor{darkblue}{\textbf{Beta-cutoff location}} $\quad\to\quad$ Occurs at a MAX node when $\beta \le \alpha$.
  \item \textcolor{darkblue}{\textbf{Time complexity of Alpha-Beta pruning with perfect ordering}} $\quad\to\quad$ $O(b^{d/2})$ (doubles solvable depth).
  \item \textcolor{darkblue}{\textbf{Does pruning change the optimal minimax decision?}} $\quad\to\quad$ No, it returns the identical move.

\section*{Chapter 8: First-Order Logic \& Knowledge Representation}

\begin{enumerate}[label=\arabic*., leftmargin=*, labelsep=0.8em, itemsep=0.35em, parsep=0.1em, topsep=0.2em]
  \item \textcolor{darkblue}{\textbf{Declarative Knowledge definition}} $\quad\to\quad$ Static facts and assertions representing concepts ("knowing what").
  \item \textcolor{darkblue}{\textbf{Procedural Knowledge definition}} $\quad\to\quad$ Steps, rules, and procedures to perform tasks ("knowing how").
  \item \textcolor{darkblue}{\textbf{Production System components}} $\quad\to\quad$ Production rules (Rule Base), Working Memory, and Inference Engine.
  \item \textcolor{darkblue}{\textbf{Production System rule format}} $\quad\to\quad$ IF $\langle$Condition$\rangle$ $\rightarrow$ THEN $\langle$Action$\rangle$.
  \item \textcolor{darkblue}{\textbf{Recognize-Act cycle steps}} $\quad\to\quad$ Match rules, resolve conflict, execute action.
  \item \textcolor{darkblue}{\textbf{Horn Clause definition}} $\quad\to\quad$ A clause containing at most one positive literal.
  \item \textcolor{darkblue}{\textbf{Definite Clause definition}} $\quad\to\quad$ Horn clause with exactly one positive literal.
  \item \textcolor{darkblue}{\textbf{Implication conversion to Horn Clause ($p \rightarrow q$)}} $\quad\to\quad$ $\neg p \lor q$ (definite clause).
  \item \textcolor{darkblue}{\textbf{Fact in Horn Clauses}} $\quad\to\quad$ Horn clause with one positive, zero negative literals.
  \item \textcolor{darkblue}{\textbf{Goal Clause in Horn Clauses}} $\quad\to\quad$ Horn clause with zero positive literals.
  \item \textcolor{darkblue}{\textbf{Skolemisation definition}} $\quad\to\quad$ Eliminating existential quantifiers ($\exists$) by substituting concrete terms.
  \item \textcolor{darkblue}{\textbf{Skolem Constant substitution}} $\quad\to\quad$ $\exists x P(x) \Longrightarrow P(A)$ (no universal scope).
  \item \textcolor{darkblue}{\textbf{Skolem Function substitution}} $\quad\to\quad$ $\forall x\exists y P(x,y)\Longrightarrow\forall x P(x,F(x))$ (universal scope).
  \item \textcolor{darkblue}{\textbf{Prolog programming paradigm}} $\quad\to\quad$ Declarative logic programming language based on Horn clauses.
  \item \textcolor{darkblue}{\textbf{Prolog base case for factorial of 0}} $\quad\to\quad$ `factorial(0, 1).`
  \item \textcolor{darkblue}{\textbf{Prolog mod operator in GCD recursive rule}} $\quad\to\quad$ `R is X mod Y.`
  \item \textcolor{darkblue}{\textbf{FOL to CNF conversion Step 1}} $\quad\to\quad$ Eliminate implications/biconditionals ($A \rightarrow B \Rightarrow \neg A \lor B$).
  \item \textcolor{darkblue}{\textbf{Move negations inward FOL rule}} $\quad\to\quad$ $\neg \forall x P(x) \Longrightarrow \exists x \neg P(x)$.
  \item \textcolor{darkblue}{\textbf{Standardize variables step}} $\quad\to\quad$ Renaming variables to ensure unique names per quantifier.
  \item \textcolor{darkblue}{\textbf{Standard CNF format}} $\quad\to\quad$ Conjunction of disjunction of literals (conjunction of clauses).
  \item \textcolor{darkblue}{\textbf{Resolution Refutation proof method}} $\quad\to\quad$ Negate target goal, resolve with KB clauses to derive $\Box$.
  \item \textcolor{darkblue}{\textbf{Resolvent of $P(x) \lor Q(x)$ and $\neg P(A)$ using $\{x/A\}$}} $\quad\to\quad$ $Q(A)$.
  \item \textcolor{darkblue}{\textbf{Semantic Network representation}} $\quad\to\quad$ Concept nodes connected by directed, labeled relation edges.
  \item \textcolor{darkblue}{\textbf{Frame-based representation attributes and values}} $\quad\to\quad$ Slots (attributes) and fillers (values).
  \item \textcolor{darkblue}{\textbf{Inheritance in Frames}} $\quad\to\quad$ Subclasses inherit default slot values from superclasses.
  \item \textcolor{darkblue}{\textbf{Fuzzy Set membership values}} $\quad\to\quad$ Real numbers representing membership degree in $[0, 1]$.
  \item \textcolor{darkblue}{\textbf{Fuzzy Union operator ($\cup$)}} $\quad\to\quad$ $\mu_{A \cup B}(x) = \max(\mu_A(x), \mu_B(x))$.
  \item \textcolor{darkblue}{\textbf{Fuzzy Intersection operator ($\cap$)}} $\quad\to\quad$ $\mu_{A \cap B}(x) = \min(\mu_A(x), \mu_B(x))$.

  \item \textcolor{darkblue}{\textbf{Crossover operator in Genetic Algorithms}} $\quad\to\quad$ Swapping genetic segments between two parent chromosomes.
  \item \textcolor{darkblue}{\textbf{Mutation operator in Genetic Algorithms}} $\quad\to\quad$ Randomly flipping gene values to maintain diversity.
\end{enumerate}
